# Stage4b is a HOST TEST TOOL. It is not a second player, shared library, APK
# dependency or runtime audio backend. Existing lmg_automix DSP is linked as-is.
if(ANDROID)
  message(FATAL_ERROR "The offline PCM bench must not be built into Android")
endif()
if(NOT TARGET lmg_automix)
  message(FATAL_ERROR "Build the offline bench through native/automix")
endif()
add_library(lmg_automix_pcm_support STATIC pcm_support.cpp)
target_include_directories(lmg_automix_pcm_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(lmg_automix_pcm_support PUBLIC cxx_std_17)
add_library(lmg_automix_pcm_offline STATIC pcm_bench.cpp pcm_fixtures.cpp)
target_link_libraries(lmg_automix_pcm_offline PUBLIC lmg_automix lmg_automix_pcm_support)
target_compile_features(lmg_automix_pcm_offline PUBLIC cxx_std_17)
foreach(target lmg_automix_pcm_support lmg_automix_pcm_offline)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  if(LMG_AUTOMIX_SANITIZERS)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
add_executable(lmg_automix_pcm_bench pcm_demo.cpp)
target_link_libraries(lmg_automix_pcm_bench PRIVATE lmg_automix_pcm_offline)
add_executable(lmg_automix_pcm_tests pcm_tests.cpp)
target_link_libraries(lmg_automix_pcm_tests PRIVATE lmg_automix_pcm_offline)
add_executable(lmg_automix_pcm_support_tests pcm_support_test.cpp)
target_link_libraries(lmg_automix_pcm_support_tests PRIVATE lmg_automix_pcm_support)
foreach(target lmg_automix_pcm_bench lmg_automix_pcm_tests lmg_automix_pcm_support_tests)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endforeach()
add_test(NAME automix_pcm_utils COMMAND lmg_automix_pcm_support_tests)
add_test(NAME automix_pcm_transport COMMAND lmg_automix_pcm_tests transport)
add_test(NAME automix_pcm_latency_tails COMMAND lmg_automix_pcm_tests latency-tails)
add_test(NAME automix_pcm_schedule COMMAND lmg_automix_pcm_tests schedule "${LMG_AUTOMIX_STYLE_CATALOG}")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME automix_pcm_report_guard COMMAND ${Python3_EXECUTABLE}
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../scripts/automix_pcm_report_test.py")
set_tests_properties(automix_pcm_utils automix_pcm_transport automix_pcm_latency_tails
  automix_pcm_schedule automix_pcm_report_guard PROPERTIES LABELS "offline_pcm;stage4b" TIMEOUT 300)

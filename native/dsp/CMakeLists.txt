cmake_minimum_required(VERSION 3.22.1)
project(LmgPlayerDsp LANGUAGES C CXX)

option(LMG_PLAYER_DSP_TESTS "Build PCM adapter tests" OFF)
add_subdirectory(vendor/lmg_dsp)
add_library(lmg_dsp_pcm STATIC pcm_processor.cpp)
target_include_directories(lmg_dsp_pcm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(lmg_dsp_pcm PUBLIC LMG::dsp)
target_compile_features(lmg_dsp_pcm PUBLIC cxx_std_17)
set_target_properties(lmg_dsp_pcm PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(lmg_dsp_pcm PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)

if(ANDROID OR LMG_PLAYER_DSP_TESTS)
    add_library(lmg_player_dsp SHARED dsp_jni.cpp)
    target_link_libraries(lmg_player_dsp PRIVATE lmg_dsp_pcm)
    target_compile_options(lmg_player_dsp PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
    if(ANDROID)
        # Audio deadlines apply to debug APKs too. Keep strict FP semantics while
        # optimizing the sample loop and its conversion/JNI adapters.
        foreach(dsp_target lmg_dsp lmg_dsp_pcm lmg_player_dsp)
            target_compile_options(${dsp_target} PRIVATE -O2)
        endforeach()
    endif()
    if(NOT ANDROID)
        find_package(JNI REQUIRED)
        target_include_directories(lmg_player_dsp PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
endif()

if(LMG_PLAYER_DSP_TESTS)
    enable_testing()
    add_executable(lmg_dsp_pcm_tests tests/pcm_processor_test.cpp)
    target_link_libraries(lmg_dsp_pcm_tests PRIVATE lmg_dsp_pcm)
    add_test(NAME lmg_dsp_pcm_tests COMMAND lmg_dsp_pcm_tests)
endif()
